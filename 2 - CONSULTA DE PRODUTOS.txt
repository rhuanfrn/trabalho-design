//2 - CONSULTA DE PRODUTOS

#include <stdio.h>
#include <string.h>

int main() {
    char nomes[100][50];
    float precos[100];
    int quantidades[100];
    int total = 0;
    int opcaoConsulta;

    printf("\n----- CONSULTA DE PRODUTOS -----\n");
    printf("1 - Listar todos os produtos\n");
    printf("2 - Pesquisar produto pelo nome\n");
    printf("\nEscolha uma opção: ");
    scanf("%d", &opcaoConsulta);

    if (opcaoConsulta == 1){
        if (total == 0){
            printf("\nNenhum produto cadastrado!\n");
        }

        for (int i = 0; i < total; i++){
            printf("\n----- PRODUTO %d -----\n", i + 1);
            printf("Nome: %s\n", nomes[i]);
            printf("Preço: R$ %.2f\n", precos[i]);
            printf("Quantidade: %d\n", quantidades[i]);
        }
    }

    else if (opcaoConsulta == 2){
        char busca[50];
        int encontrado = 0;

        printf("\nDigite o nome do produto: ");
        scanf(" %49[^\n]", busca);

        for (int i = 0; i < total; i++){

            if (strcmp(nomes[i], busca) == 0){
                printf("\n----- PRODUTO ENCONTRADO -----\n");
                printf("Nome: %s\n", nomes[i]);
                printf("Preço: R$ %.2f\n", precos[i]);
                printf("Quantidade: %d\n", quantidades[i]);

                encontrado = 1;
                break;
            }
        }

        if (encontrado == 0){
            printf("\nProduto não encontrado!\n");
        }

    }

    else {
        printf("\nOpção inválida!\n");
    }



    return 0;
}
